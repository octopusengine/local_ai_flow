# Memory MCP example: entities persist in a project-local JSONL file across separate tool calls.
# The server is configured by mcp/memory_server.json and uses only disposable test data.

python3 cli_ollama.py --project project_test
python3 cli_ollama.py --selector mcp_memory
python3 cli_ollama.py --clear_log

# 1. Discover the external server's tools.
python3 cli_mcp.py --server-config mcp/memory_server.json --list

# 2. Create a small knowledge-graph entity.
#    Re-running this flow is safe: the reference server ignores an entity name that already exists.
python3 cli_mcp.py --server-config mcp/memory_server.json --function create_entities --args '{"entities":[{"name":"mcp_flow_demo","entityType":"demo","observations":["Created by flow_mcp_memory.txt.","The Memory MCP server stores this entity in a project-local JSONL file."]}]}'

# 3. Search and then open the stored entity in separate MCP calls.
python3 cli_mcp.py --server-config mcp/memory_server.json --function search_nodes --args '{"query":"mcp_flow_demo"}'
python3 cli_mcp.py --server-config mcp/memory_server.json --function open_nodes --args '{"names":["mcp_flow_demo"]}' --out memory_mcp_read.json
