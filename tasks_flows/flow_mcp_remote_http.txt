# Public remote MCP example: read-only Website Spec server over Streamable HTTP.
# The configuration contains no account, token, or secret.

python3 cli_ollama.py --project project_test
python3 cli_ollama.py --selector mcp_remote_http
python3 cli_ollama.py --clear_log

# 1. Discover the public remote server's tools.
python3 cli_mcp.py --server-config mcp/website_spec_remote_server.json --list --timeout 45

# 2. Call one documented parameterless, read-only tool and save its returned text.
python3 cli_mcp.py --server-config mcp/website_spec_remote_server.json --function get_categories --timeout 45 --json --no-db --out remote_mcp_categories.txt
