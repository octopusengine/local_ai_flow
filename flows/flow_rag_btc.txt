# python .\runner.py flow_rag_btc.txt
# Předpoklad: rag_wiki/data/wiki_btc.db už byla naplněna z rag_wiki/src/btc.

python3 cli_ollama.py --project project_example
python3 cli_ollama.py --selector rag_btc
python3 cli_ollama.py --clear_log

# 1) Stejný dotaz bez RAG: výchozí task_base a obecné znalosti modelu.
python cli_ollama.py --type task_base.json --sc-cz --input "Co to je těžba bitcoinu?" --out btc_mining_plain.txt

# 2) Stejný dotaz s lokálním kontextem vyhledaným z wiki_btc.db. Pro FTS5
# používáme normalizovanou frázi z dokumentace (české skloňování jinak neřeší).
python cli_vector.py --db btc context "těžba bitcoinů" --mode text -k 3 --out wiki_btc_mining_context.txt
python cli_tool.py --clr
python cli_tool.py --add "RAG: wiki_btc — těžba bitcoinu" wiki_btc_mining_context.txt
python cli_ollama.py --type task_base.json --sc-cz --input "Co to je těžba bitcoinu?" --rules "Odpovídej pouze podle dodaného RAG kontextu. Neznámé informace nedoplňuj. Na konci uveď zdroj v hranatých závorkách." --context "RAG: wiki_btc" tools_context.txt --out btc_mining_rag.txt

# 3) Negativní test: wiki_btc neobsahuje spalovací motor, takže odpověď musí
# přiznat chybějící podklady namísto použití obecných znalostí modelu.
python cli_vector.py --db btc context "co to je spalovací motor" --mode text -k 3 --out wiki_btc_engine_context.txt
python cli_tool.py --clr
python cli_tool.py --add "RAG: wiki_btc — spalovací motor" wiki_btc_engine_context.txt
python cli_ollama.py --type task_base.json --sc-cz --input "Co to je spalovací motor?" --rules "Odpovídej pouze podle dodaného RAG kontextu. Pokud neobsahuje zdrojový chunk k dotazu, odpověz přesně: Ve wiki_btc jsem pro tento dotaz nenašel podklady. Nevymýšlej doplnění." --context "RAG: wiki_btc" tools_context.txt --out btc_engine_rag.txt

# 4) Srovnání první odpovědi bez RAG a druhé odpovědi podložené wiki_btc.
python cli_tool.py --clr
python cli_tool.py --add "Odpověď bez RAG" btc_mining_plain.txt
python cli_tool.py --add "Odpověď s RAG wiki_btc" btc_mining_rag.txt
python cli_ollama.py --type task_base.json --sc-cz --sc compare --input "Porovnej odpověď bez RAG a odpověď s RAG na otázku o těžbě bitcoinu. Zaměř se na věcnou přesnost, dohledatelnost zdrojů, konkrétnost a případné nepodložené informace. Stručně shrň rozdíly." --context "Porovnávané odpovědi" tools_context.txt --out btc_mining_compare.txt
