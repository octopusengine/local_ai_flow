# Eight-step Czech prompt chain using the current slash-command and context flow.
# Unlike flow_merge5_prompts.txt, the preceding result is reference context,
# while the next request remains a separate current input.  This avoids mixing
# instructions with source material through --merge.  The single temporary.txt
# file holds only the immediately preceding result; the full chain is in the DB.
#
# python3 runner.py flow_merge8_prompts.txt

python3 cli_ollama.py --project project_example
python3 cli_ollama.py --selector merge8_prompts
python3 cli_ollama.py --clear_log

# 1. Generate an initial set of discussion topics.
python3 cli_ollama.py --type task_base.json --sc-cz --sc brainstorm --sc list --input "Navrhni pět témat k diskusi o tom, jak má malý tým zavádět novou funkci do aplikace." --seed 42 --out temporary.txt

# 2. Use the topics as data and create a practical plan around one of them.
python3 cli_ollama.py --type task_base.json --sc-cz --sc plan --sc steps --context temporary.txt --input "Vyber jedno téma, které není první v seznamu, a vytvoř pro něj proveditelný plán na čtyři týdny." --seed 42 --out temporary.txt

# 3. Explain the chosen plan through a memorable analogy.
python3 cli_ollama.py --type task_base.json --sc-cz --sc explain --sc analogy --context temporary.txt --input "Vysvětli hlavní myšlenku tohoto plánu pomocí analogie s úpadkem Římské říše. Zachovej věcný a nepatetický tón." --seed 42 --out temporary.txt

# 4. Condense the analogy into a short executive summary.
python3 cli_ollama.py --type task_base.json --sc-cz --sc summarize --sc brief --context temporary.txt --input "Shrň nejdůležitější myšlenky do dvou krátkých odstavců pro vedoucího týmu." --seed 42 --out temporary.txt

# 5. Review the summary critically and make risks explicit.
python3 cli_ollama.py --type task_base.json --sc-cz --sc explain --sc critic --sc bulletpoints --context temporary.txt --input "Konstruktivně zpochybni tuto tezi: pojmenuj slabiny, rizika a mezery v argumentaci a ke každému bodu navrhni zlepšení." --seed 42 --out temporary.txt

# 6. Turn the critique into a prioritized corrective plan.
python3 cli_ollama.py --type task_base.json --sc-cz --sc plan --sc steps --context temporary.txt --input "Připrav prioritizovaný plán nápravy. Uveď první tři kroky, odpovědnou roli a očekávaný výsledek každého kroku." --seed 42 --out temporary.txt

# 7. Communicate the corrective plan to the team as an email.
python3 cli_ollama.py --type task_base.json --sc-cz --sc email --context temporary.txt --input "Napiš stručný interní e-mail týmu, který představí tento plán, požádá o spolupráci a uvede nejbližší další krok." --seed 42 --out temporary.txt

# 8. Make the final email natural and concise without changing its purpose.
python3 cli_ollama.py --type task_base.json --sc-cz --sc rewrite --sc grammar --sc human --sc brief --context temporary.txt --input "Přepracuj e-mail pro přirozený, stručný a profesionální tón. Zachovej jeho význam, konkrétní požadavek i nejbližší další krok." --seed 42 --out temporary.txt
