# Before the first run, Ollama must provide the embedding model from cli_vector.json:
#   ollama pull embeddinggemma
# This flow uses embeddings stored in rag_wiki/data/wiki_btc.db.

python3 cli_ollama.py --project project_example
python3 cli_ollama.py --selector vector_btc_en
python3 cli_ollama.py --clear_log

# Database maintenance does not belong in a query flow. After changing sources,
# use James → RAG → ingest → u; overwrite (o) is for a changed PDF extractor or
# chunking. For a different embedding model, create a new wiki database.
# python cli_vector.py --db btc ingest --reindex

# Incremental ingest: also reads rag_wiki/src/btc/web_src.json and contacts
# declared pages one by one. Unchanged local files and unchanged page text are skipped.
python cli_vector.py --db btc ingest
python cli_vector.py --db btc inspect
python cli_vector.py --db btc verify

# The query uses different wording from the source headings, demonstrating
# semantic embedding similarity instead of an exact FTS5 token match.
python cli_vector.py --db btc search "How are new bitcoins created and why does mining consume energy?" --mode vector -k 3
python cli_vector.py --db btc context "How are new bitcoins created and why does mining consume energy?" --mode vector -k 3 --out wiki_btc_vector_context_en.txt

python cli_tool.py --clr
python cli_tool.py --add "RAG: wiki_btc — vector search" wiki_btc_vector_context_en.txt
python cli_ollama.py --type task_base.json --sc-en --input "How does Bitcoin mining work and why does it consume energy?" --rules "Answer only from the supplied RAG context. Do not add unknown information. End with the used source in square brackets." --context "RAG: wiki_btc (vector)" tools_context.txt --out btc_vector_answer_en.txt
