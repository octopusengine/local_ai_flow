# Dávkové OCR: z PROJECT/src vybere .png/.jpg/.jpeg soubory a jejich
# přepis uloží do PROJECT/dest jako NAME.png.txt nebo NAME.jpg.txt.

$batch_in = "src"
$batch_out = "dest"

python3 cli_ollama.py --project project_example
python3 cli_ollama.py --selector batch_ocr
python3 cli_ollama.py --clear_log

python3 cli_tool.py --echo "[ --- OCR batch images --- ]"
python3 cli_tool.py --batch-img

@for IMAGE in $batch_list
    python3 cli_ollama.py --type task_ocr.json --sc /ocr --in $batch_in/$IMAGE --out $batch_out/$IMAGE.ocr.txt
@endfor
